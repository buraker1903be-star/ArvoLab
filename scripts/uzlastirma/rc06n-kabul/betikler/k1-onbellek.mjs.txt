// RC06n K1 (PR #106 sonrası aday): iki kullanıcı, DOI'li/DOI'siz, soğuk/sıcak doğrulama önbelleği (test düzeneği).
// Kullanım: node k1-onbellek.mjs <adres> <posta.jsonl> <dizin.jsonl> <çıktı dizini> <senaryo.json>
// Her adım: kullanıcı Kaynakça → "Manuel ekle" → yapıştır → "Yapıştırdığımı denetle". Sonuç, uygulamanın
// kendi yazdığı citation_checks satırından (cross_check.academicVerification) okunur; dizine giden istek
// sayısı yakalayıcının günlüğünden (dizin.jsonl) adım başına sayılır; önbellek tablosu her adımdan sonra alınır.
import fs from "node:fs";
import { chromium, db, kullaniciAc, satirSay, satirlar } from "./ortak.mjs";

const [URL0, POSTA, DIZIN, OUT, SENARYO] = process.argv.slice(2);
const senaryo = JSON.parse(fs.readFileSync(SENARYO, "utf8"));
const b = await chromium.launch();
const kisiler = {};
for (const k of senaryo.kullanicilar) {
  kisiler[k] = await kullaniciAc(b, URL0, POSTA, OUT, `n1${k.toLowerCase()}`);
  // Kaynakça editördeki metinden alınır (çalışmaya bağlı denetim gövdesiz koşmuyor): RC06k z1 ile aynı yol.
  const s = kisiler[k].sayfa;
  await s.goto(`/dashboard/editor/${kisiler[k].proje}/write`, { timeout: 120000 });
  await s.waitForSelector(".ProseMirror", { timeout: 120000 });
  await s.locator(".ProseMirror").click();
  await s.keyboard.press("Control+End");
  await s.keyboard.type("Bu çalışma kaynak doğrulama önbelleğinin kabulü için yazılmış sentetik bir metindir. ");
  await s.keyboard.press("Enter");
  await s.keyboard.type("## Kaynakça");
  // Yavaş yazım: B adımında hızlı yazım bir harfi önceki satıra kaydırdı ("Sohret" → "ohret"), betik kaynaklı.
  for (const kunye of senaryo.kaynakca[k]) { await s.keyboard.press("Enter"); await s.waitForTimeout(400); await s.keyboard.type(kunye, { delay: 30 }); await s.waitForTimeout(400); }
  await s.waitForFunction(() => /Kaydedildi/.test(document.body.innerText), null, { timeout: 60000 });
  await s.waitForTimeout(2500);
}
const adimlar = [];
for (const adim of senaryo.adimlar) {
  const kisi = kisiler[adim.kim];
  const s = kisi.sayfa;
  const onceSatir = Number(db(`select count(*) from citation_checks where created_by='${kisi.kimlik}'`));
  const onceDizin = satirSay(DIZIN);
  await s.goto(`/dashboard/citations?calisma=${kisi.proje}`);
  try {
    // İlk denetimden sonra sayfadaki düğme "Yeniden denetle" olabiliyor: ikisini de kabul et.
    await s.getByRole("button", { name: /^Kaynakçamı denetle$|^Sonuçları aç$/ }).first().click({ timeout: 30000 });
    await s.waitForTimeout(800);
    await s.locator('[role="dialog"]').getByRole("button", { name: /^Kaynakçamı denetle$/ }).click({ timeout: 30000 });
  } catch (h) {
    await s.screenshot({ path: `${OUT}/hata-${adim.ad}.png` });
    console.log("HATA", adim.ad, String(h.message).split("\n")[0]);
    console.log((await s.locator("main").innerText().catch(() => "")).replace(/\s+/g, " ").slice(0, 900));
    throw h;
  }
  for (let i = 0; i < 240; i++) {
    if (Number(db(`select count(*) from citation_checks where created_by='${kisi.kimlik}'`)) > onceSatir) break;
    await s.waitForTimeout(500);
  }
  const ham = db(`select cross_check->'academicVerification' from citation_checks where created_by='${kisi.kimlik}' order by created_at desc limit 1`);
  const dogrulama = JSON.parse(ham || "[]").map((v) => ({ kunye: v.reference, durum: v.status, enIyiDoi: v.bestMatch?.doi ?? null, enIyiBaslik: v.bestMatch?.title ?? null, enIyiYazarlar: (v.bestMatch?.authors ?? []).slice(0, 3), istenenDoi: v.istenenDoi ?? null }));
  await s.waitForTimeout(1500);
  const ekran = (await s.locator(".kunye-liste").innerText().catch(() => "")).replace(/\s+/g, " ").slice(0, 1200);
  const dizin = satirlar(DIZIN, onceDizin);
  const onbellek = db(`select json_agg(json_build_object('anahtar',anahtar,'durum',durum,'enIyiDoi',en_iyi->>'doi') order by anahtar) from kaynak_dogrulama_onbellegi`);
  const kayit = { ad: adim.ad, kim: adim.kim, kullanici: kisi.kimlik, dogrulama, dizinIstekleri: dizin.length, dizinDurumlari: dizin.map((d) => `${d.alan}${d.yol.split("?")[0]} ${d.durum}`), onbellek: JSON.parse(onbellek || "[]"), ekran };
  adimlar.push(kayit);
  console.log(`ADIM ${adim.ad} (${adim.kim}) dizin=${dizin.length}`);
  for (const d of dogrulama) console.log(`   ${d.durum.padEnd(15)} enIyi=${d.enIyiDoi} istenen=${d.istenenDoi} | ${d.kunye.slice(0, 70)}`);
}
fs.writeFileSync(`${OUT}/k1-onbellek.json`, JSON.stringify({ kullanicilar: Object.fromEntries(Object.entries(kisiler).map(([k, v]) => [k, { kimlik: v.kimlik, proje: v.proje }])), adimlar, konsol: Object.fromEntries(Object.entries(kisiler).map(([k, v]) => [k, v.konsol.slice(-10)])) }, null, 2));
await b.close();
