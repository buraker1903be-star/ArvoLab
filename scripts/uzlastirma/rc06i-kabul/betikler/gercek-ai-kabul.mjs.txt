async function auth(yol, { yontem = "POST", govde, belirtec = ANON, apikey = ANON } = {}) {
  const r = await fetch(`${GECIT}/auth/v1${yol}`, {
    method: yontem,
    headers: { apikey, Authorization: `Bearer ${belirtec}`, "Content-Type": "application/json" },
    body: govde ? JSON.stringify(govde) : undefined,
  });
  const t = await r.text();
  let json = null;
  try {
    json = JSON.parse(t);
  } catch {}
  return { durum: r.status, json };
}

    return { _ham: r.durum };

// Alıntı/sayfa doğruluğu: her alıntı belirtilen sayfanın metninde birebir (boşluk/tırnak normalize) geçmeli.
function alintiDenetle(alintilar = []) {
  return alintilar.map((a) => {
    const sayfa = Number(a.sayfa);
    const metin = norm(a.metin);
    const sayfadaVar = sayfa >= 1 && sayfa <= SAYFALAR.length && SAYFA_METNI[sayfa - 1].includes(metin);
    const baskaSayfada = !sayfadaVar ? SAYFA_METNI.findIndex((t) => t.includes(metin)) + 1 : 0;
    return { sayfa, sayfadaVar, baskaSayfada, uzunluk: metin.length };
  });
}

  return y;
}

// ── 6) ai_assistant_runs ve veri bütünlüğü ───────────────────────────────
{
  const satirlar = psql(`select status, coalesce(reject_reason,'-'), coalesce(model,'-'), capability, (project_id='${PROJE}')::text, (context is not null)::text, (output is not null)::text, coalesce(output_chars,0) from public.ai_assistant_runs where user_id='${K.id}' order by created_at`).split("\n").filter(Boolean);
  const kosular = satirlar.map((s) => s.split("|"));
  const tamam = kosular.filter((k) => k[0] === "completed");
  const red = kosular.filter((k) => k[0] === "rejected");
  const hatali = kosular.filter((k) => k[0] === "failed");
  const beklenenKosu = yanitlar.length + 1; // modelli çağrılar + F1
  kayit(
    "K1",

    `kayıt = modelli çağrı (${yanitlar.length}) + F1 (1) = ${beklenenKosu}`,
    `kayıt ${kosular.length} (completed ${tamam.length}, rejected ${red.length}, failed ${hatali.length})`,
    kosular.length === beklenenKosu ? "PASS" : "FAIL",
  );
  const alanlar = kosular.every((k) => k[3] === "kaynak" && k[4] === "true" && k[5] === "true") &&

    hatali.every((k) => k[6] === "false");

  // İçerik değiştirilemez (guard_ai_run_update): kullanıcı kendi kaydının çıktısını PostgREST'ten yazamaz.
  const id = psql(`select id from public.ai_assistant_runs where user_id='${K.id}' and status='completed' order by created_at limit 1`);
  const ciktiOnce = psql(`select md5(coalesce(output,'')) from public.ai_assistant_runs where id='${id}'`);
  const r = await fetch(`${GECIT}/rest/v1/ai_assistant_runs?id=eq.${id}`, {
    method: "PATCH",
    headers: { apikey: ANON, Authorization: `Bearer ${K.erisim}`, "Content-Type": "application/json", Prefer: "return=minimal" },
    body: JSON.stringify({ output: "degistirildi" }),
  });

  kayit("K3", "Asistan kaydının çıktısı sonradan değiştirilemez", "çıktı aynı kalır", `PATCH ${r.status} · çıktı aynı ${ciktiOnce === ciktiSonra}`, ciktiOnce === ciktiSonra ? "PASS" : "FAIL");
  // Anahtar sızıntısı: kayıtlarda, günlüklerde sağlayıcı anahtarı biçimi yok.
  const dbSizinti = Number(psql(`select count(*) from public.ai_assistant_runs where coalesce(output,'') || coalesce(context::text,'') like '%sk-ant%'`));
