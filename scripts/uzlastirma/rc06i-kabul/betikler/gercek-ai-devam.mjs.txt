  });
}

  // Anahtar sızıntısı: kayıtlarda, günlüklerde sağlayıcı anahtarı biçimi yok.
  const dbSizinti = Number(psql(`select count(*) from public.ai_assistant_runs where coalesce(output,'') || coalesce(context::text,'') like '%sk-ant%'`));
