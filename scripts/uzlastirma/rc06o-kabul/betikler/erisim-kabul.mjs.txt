async function hesap(ad, rol) {
  const adres = `${ad}-${ek}@yerel.test`, parola = `Kq7!${crypto.randomBytes(9).toString("base64url")}`;
  const r = await yonetici("/admin/users", { email: adres, email_confirm: true });

  // G1 tetikleyicisi onayda şifreyi siler; şifre ayrı güncellemeyle konur (kimlik-kabul.mjs ile aynı).

  if (g.durum >= 300) throw new Error(`şifre konamadı ${g.durum}`);
